from pathlib import Path
import json,sys,subprocess,hashlib,base64
repo=Path(sys.argv[1]);request=json.load(sys.stdin)
assert subprocess.run(['git','rev-parse','--show-toplevel'],cwd=repo,capture_output=True,text=True,check=True).stdout.strip()==str(repo)
assert subprocess.run(['nvidia-smi','--query-gpu=uuid','--format=csv,noheader'],capture_output=True,text=True,check=True).stdout.splitlines()==['GPU-44039938-fd82-41d2-fefd-de71514e2fac']
root=repo/'experiments_iclr/postsubmission_20260930/hgb_acm_confirmation_root_v1';root.mkdir(exist_ok=False)
for row in request:
 assert Path(row['name']).name==row['name'];b=base64.b64decode(row['data']);assert len(b)==row['bytes'] and hashlib.sha256(b).hexdigest()==row['sha256']
 (root/row['name']).write_bytes(b)
r=subprocess.run(['/usr/bin/python3','-I','-S','-B',str(root/'acquire_and_inspect_acm.py')],cwd=repo,capture_output=True,text=True,timeout=240)
print(json.dumps(dict(exit_code=r.returncode,stdout=r.stdout,stderr=r.stderr)));sys.exit(r.returncode)
