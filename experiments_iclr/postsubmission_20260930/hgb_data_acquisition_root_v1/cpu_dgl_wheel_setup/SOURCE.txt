import pathlib,json,subprocess,urllib.request,zipfile,hashlib,os
repo=pathlib.Path('/home/jovyan/shares/SR003.nfs2/GENATATOR_PIPELINE/diploma_yandex_graphs');assert pathlib.Path.cwd()==repo
assert subprocess.run(['nvidia-smi','--query-gpu=uuid','--format=csv,noheader'],capture_output=True,text=True,check=True).stdout.splitlines()==['GPU-44039938-fd82-41d2-fefd-de71514e2fac']
base=repo/'.gnnm_runtime/hgb_dgl_cpu_v1';site=base/'site';assert site.is_dir() and not any(site.iterdir())
with urllib.request.urlopen('https://pypi.org/pypi/dgl/1.1.3/json',timeout=25) as r:metadata=json.load(r)
choices=[r for r in metadata['urls'] if r['filename']=='dgl-1.1.3-cp311-cp311-manylinux1_x86_64.whl'];assert len(choices)==1
row=choices[0]
with urllib.request.urlopen(row['url'],timeout=30) as r:data=r.read(20000000)
assert len(data)==row['size'] and hashlib.sha256(data).hexdigest()==row['digests']['sha256']
wheel=base/row['filename'];assert not wheel.exists();wheel.write_bytes(data)
with zipfile.ZipFile(wheel) as z:
 for item in z.infolist():
  path=pathlib.PurePosixPath(item.filename);assert not path.is_absolute() and '..' not in path.parts
 z.extractall(site)
env=os.environ.copy();env.update(PYTHONPATH=str(site),CUDA_VISIBLE_DEVICES='',DGLBACKEND='pytorch',PYTHONDONTWRITEBYTECODE='1')
check=subprocess.run([str(repo/'.venv/bin/python'),'-B','-c','import torch,dgl,json;print(json.dumps(dict(torch=torch.__version__,dgl=dgl.__version__,dgl_file=dgl.__file__)))'],cwd=repo,env=env,capture_output=True,text=True,timeout=40)
receipt=dict(status='qualified_CPU_DGL_import' if check.returncode==0 else 'CPU_DGL_import_failed',wheel_filename=row['filename'],wheel_sha256=row['digests']['sha256'],wheel_bytes=len(data),package_site=str(site),native_env_modified=False,GPU_models_executed=False,check_exit_code=check.returncode,stdout=check.stdout,stderr=check.stderr)
with (base/'CPU_WHEEL_SETUP.json').open('x') as h:json.dump(receipt,h,indent=2);h.write('\n')
print(json.dumps(receipt))
