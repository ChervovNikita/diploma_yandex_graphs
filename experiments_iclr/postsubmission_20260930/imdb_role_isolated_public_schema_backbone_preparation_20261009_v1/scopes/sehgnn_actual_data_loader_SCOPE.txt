0269     def load_nodes(self):
0270         """
0271         return nodes dict
0272             total: total number of nodes
0273             count: a dict of int, number of nodes for each type
0274             attr: a dict of np.array (or None), attribute matrices for each type of nodes
0275             shift: node_id shift for each type. You can get the id range of a type by
0276                         [ shift[node_type], shift[node_type]+count[node_type] )
0277         """
0278         nodes = {'total':0, 'count':Counter(), 'attr':{}, 'shift':{}}
0279         with open(os.path.join(self.path, 'node.dat'), 'r', encoding='utf-8') as f:
0280             for line in f:
0281                 th = line.split('\t')
0282                 if len(th) == 4:
0283                     # Then this line of node has attribute
0284                     node_id, node_name, node_type, node_attr = th
0285                     node_id = int(node_id)
0286                     node_type = int(node_type)
0287                     node_attr = list(map(float, node_attr.split(',')))
0288                     nodes['count'][node_type] += 1
0289                     nodes['attr'][node_id] = node_attr
0290                     nodes['total'] += 1
0291                 elif len(th) == 3:
0292                     # Then this line of node doesn't have attribute
0293                     node_id, node_name, node_type = th
0294                     node_id = int(node_id)
0295                     node_type = int(node_type)
0296                     nodes['count'][node_type] += 1
0297                     nodes['total'] += 1
0298                 else:
0299                     raise Exception("Too few information to parse!")
0300         shift = 0
0301         attr = {}
0302         for i in range(len(nodes['count'])):
0303             nodes['shift'][i] = shift
0304             if shift in nodes['attr']:
0305                 mat = []
0306                 for j in range(shift, shift+nodes['count'][i]):
0307                     mat.append(nodes['attr'][j])
0308                 attr[i] = np.array(mat)
0309             else:
0310                 attr[i] = None
0311             shift += nodes['count'][i]
0312         nodes['attr'] = attr
0313         return nodes
