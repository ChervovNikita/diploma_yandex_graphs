0016 class Transformer(nn.Module):
0017     '''
0018         The transformer-based semantic fusion in SeHGNN.
0019     '''
0020     def __init__(self, n_channels, num_heads=1, att_drop=0., act='none'):
0021         super(Transformer, self).__init__()
0022         self.n_channels = n_channels
0023         self.num_heads = num_heads
0024         assert self.n_channels % (self.num_heads * 4) == 0
0025 
0026         self.query = nn.Linear(self.n_channels, self.n_channels//4)
0027         self.key   = nn.Linear(self.n_channels, self.n_channels//4)
0028         self.value = nn.Linear(self.n_channels, self.n_channels)
0029 
0030         self.gamma = nn.Parameter(torch.tensor([0.]))
0031         self.att_drop = nn.Dropout(att_drop)
0032         if act == 'sigmoid':
0033             self.act = torch.nn.Sigmoid()
0034         elif act == 'relu':
0035             self.act = torch.nn.ReLU()
0036         elif act == 'leaky_relu':
0037             self.act = torch.nn.LeakyReLU(0.2)
0038         elif act == 'none':
0039             self.act = lambda x: x
0040         else:
0041             assert 0, f'Unrecognized activation function {act} for class Transformer'
0042 
0043         self.reset_parameters()
0044 
0045     def reset_parameters(self):
0046         for k, v in self._modules.items():
0047             if hasattr(v, 'reset_parameters'):
0048                 v.reset_parameters()
0049         nn.init.zeros_(self.gamma)
0050 
0051     def forward(self, x, mask=None):
0052         B, M, C = x.size() # batchsize, num_metapaths, channels
0053         H = self.num_heads
0054         if mask is not None:
0055             assert mask.size() == torch.Size((B, M))
0056 
0057         f = self.query(x).view(B, M, H, -1).permute(0,2,1,3) # [B, H, M, -1]
0058         g = self.key(x).view(B, M, H, -1).permute(0,2,3,1)   # [B, H, -1, M]
0059         h = self.value(x).view(B, M, H, -1).permute(0,2,1,3) # [B, H, M, -1]
0060 
0061         beta = F.softmax(self.act(f @ g / math.sqrt(f.size(-1))), dim=-1) # [B, H, M, M(normalized)]
0062         beta = self.att_drop(beta)
0063         if mask is not None:
0064             beta = beta * mask.view(B, 1, 1, M)
0065             beta = beta / (beta.sum(-1, keepdim=True) + 1e-12)
0066 
0067         o = self.gamma * (beta @ h) # [B, H, M, -1]
0068         return o.permute(0,2,1,3).reshape((B, M, C)) + x
