def read(name, cap=524288):
    path = ROOT / name
    if not path.exists():
        return None
    assert path.resolve() == path and path.is_relative_to(ROOT) and path.is_file()
    before = path.stat()
    assert before.st_size <= cap, ('metadata receipt exceeds read cap', name, before.st_size)
    data = path.read_bytes()
    after = path.stat()
    assert (before.st_ino, before.st_size, before.st_mtime_ns) == (after.st_ino, after.st_size, after.st_mtime_ns), ('unstable receipt', name)
    assert len(data) == before.st_size
    item = dict(path=str(path), bytes=len(data), sha256=hashlib.sha256(data).hexdigest(), base64=base64.b64encode(data).decode('ascii'))
    receipts[name] = item
    return data

def observe_owned(expected):
    path = Path('/proc') / str(expected['pid']) / 'stat'
    try:
        stat = path.read_text()
    except FileNotFoundError:
        return dict(expected=expected, present=False)
    fields = stat[stat.rfind(')') + 2:].split()
    row = dict(pid=expected['pid'], start_ticks=int(fields[19]), state=fields[0], ppid=int(fields[1]), pgid=int(fields[2]), sid=int(fields[3]), RSS_bytes=int(fields[21]) * os.sysconf('SC_PAGE_SIZE'))
    matches = all(row[k] == expected[k] for k in ('pid', 'start_ticks', 'pgid', 'sid'))
    return dict(expected=expected, present=True, identity_matches=matches, observed=row)
