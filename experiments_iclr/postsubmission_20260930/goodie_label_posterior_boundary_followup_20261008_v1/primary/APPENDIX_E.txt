Appendix E Algorithm
Algorithm 1 Pseudocode for training GOODIE
1:
A graph \mathcal{G}=(\mathcal{V},\mathcal{E}) , Partially observed feature matrix \mathbf{X}^{(0)} , and train label matrix \mathbf{Y}^{(0)}
2:
Final prediction matrix \mathbf{P\in\mathrm{R}^{N\times|C|}}
3:
\hat{\mathbf{A}}_{sym}=\tilde{\mathbf{D}}^{-1/2}\tilde{\mathbf{A}}\tilde{\mathbf{D}}^{-1/2}
4:
\hat{\mathbf{Y}},\hat{\mathbf{X}},\mathcal{W} = Propagation( \textbf{Y}^{(0)},\textbf{X}^{(0)} )
5:
while Convergence do
6:
\mathbf{P},\mathbf{Z}=\textsc{Train\_GOODIE}(\hat{\mathbf{Y}},\hat{\mathbf{X}})
7:
\mathcal{L_{\text{ce}}}=\sum_{v\in\mathcal{V}_{\small{tr}}}\sum_{c\in\mathcal{C}}CE(\mathbf{p}_{v}[c])
8:
\mathcal{L_{\text{pseudo}}}=\sum_{i\in I}\frac{-1}{|P(i)|}\sum_{p\in P(i)}w_{ip}\cdot\log\frac{\exp(\mathbf{z}_{i}\cdot\mathbf{z}_{p}/\tau)}{\sum_{a\in A(i)}\exp(\mathbf{z}_{i}\cdot\mathbf{z}_{a}/\tau)}
9:
\mathcal{L_{\text{final}}}=\mathcal{L_{\text{ce}}}+\lambda\mathcal{L_{\text{pseudo}}}
10:
end while
11:
function Train_GOODIE( \hat{\mathbf{Y}},\hat{\mathbf{X}} )
12:
\mathbf{H}^{\text{LP}}=\sigma({\hat{\mathbf{A}}_{sym}}\hat{\mathbf{Y}}\mathbf{W}_{\text{LP}})
13:
\mathbf{H}^{\text{FP}}=\sigma({\hat{\mathbf{A}}_{sym}}\hat{\mathbf{X}}\mathbf{W}_{\text{HP}})
14:
for i\in\mathcal{V} do
15:
\alpha_{i,LP}=\frac{\exp({\text{LeakyReLU}(\textbf{a}^{\top}\textbf{h}_{i}^{\text{LP}})})}{\exp({\text{LeakyReLU}(\textbf{a}^{\top}\textbf{h}_{i}^{\text{LP}})})+\exp({\text{LeakyReLU}(\textbf{a}^{\top}\textbf{h}_{i}^{\text{FP}})})}
16:
\alpha_{i,FP}=\frac{\exp({\text{LeakyReLU}(\textbf{a}^{\top}\textbf{h}_{i}^{\text{FP}})})}{\exp({\text{LeakyReLU}(\textbf{a}^{\top}\textbf{h}_{i}^{\text{LP}})})+\exp({\text{LeakyReLU}(\textbf{a}^{\top}\textbf{h}_{i}^{\text{FP}})})}
17:
\mathbf{z}_{i}=\alpha_{i,\text{LP}}\mathbf{h}_{i}^{\text{LP}}+\alpha_{i,\text{FP}}\mathbf{h}_{i}^{\text{FP}}
18:
end for
19:
return \text{softmax}(\sigma({\hat{\mathbf{A}}_{sym}}\mathbf{Z}\mathbf{W}_{\text{cls}})) , Z
20:
end function
21:
function Propagation( \textbf{Y}^{(0)},\textbf{X}^{(0)} )
22:
for k=0;k<K;k=k+1 do
23:
\mathbf{Y}^{(k+1)}\leftarrow{\hat{\mathbf{A}}_{sym}}\mathbf{Y}^{(k)}
24:
\textbf{y}^{(k+1)}_{i}\leftarrow\textbf{y}^{(0)}_{i},\forall i\leq m.
25:
end for
26:
\mathcal{W}=\textsc{Weight Calculator}(\mathbf{Y}^{(K)})
27:
for k=0;k<K;k=k+1 do
28:
\mathbf{X}^{(k+1)}\leftarrow{\hat{\mathbf{A}}_{sym}}\mathbf{X}^{(k)}
29:
\mathbf{x}^{(k+1)}_{i,d}=\mathbf{x}^{(0)}_{i,d},\forall i\exists\mathcal{V}_{known,d},\forall d.
30:
end for
31:
return \mathbf{Y}^{(K)},\mathbf{X}^{(K)},\mathcal{W}
32:
end function
33:
function Weight Calculator( \hat{\mathbf{Y}} )
34:
\tilde{\mathbf{y}}=\text{max}(\text{softmax}(\hat{\mathbf{Y}}))
35:
\mathcal{W}=[\textbf{0},\cdots,\textbf{0}]^{T}
36:
for i=0;\;i<N;\;i=i+1 do
37:
for j=i;\;j<N;\;j=j+1 do
38:
if argmax(\hat{\mathbf{y}_{i}})==argmax(\hat{\mathbf{y}_{j}}) then
39:
p\leftarrow j
40:
w_{ip}=\begin{cases}1,&\text{if}\;\;i,p\in\hat{Y}_{train}\\
\tilde{y}_{p},&\text{if}\;\;i\in\hat{Y}_{train}\;\;\text{and}\;\;p\in\hat{Y}_{pseudo}\\
\tilde{y}_{i},&\text{if}\;\;i\in\hat{Y}_{pseudo}\;\;\text{and}\;\;p\in\hat{Y}_{train}\\
\tilde{y}_{i}*\tilde{y}_{p},&\text{if}\;\;i,p\in\hat{Y}_{pseudo}\end{cases}
41:
end if
42:
end for
43:
end for
44:
return \mathcal{W}
45:
end function
