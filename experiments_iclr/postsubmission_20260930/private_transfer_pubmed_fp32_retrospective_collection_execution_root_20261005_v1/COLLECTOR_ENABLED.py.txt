def digest(raw):
    return hashlib.sha256(raw).hexdigest()
def confined(path):

def checked_bytes(rel, pin=None):
    path = ROOT / rel
    confined(path)
    assert not Path(rel).is_absolute() and '..' not in Path(rel).parts
    assert ROOT.is_dir() and not ROOT.is_symlink()
    assert path.is_file() and not path.is_symlink() and path.resolve().is_relative_to(ROOT.resolve())
    assert all(not parent.is_symlink() for parent in path.parents if parent.is_relative_to(ROOT))
    assert path.stat().st_size <= OUTPUT_CAP
    raw = path.read_bytes()
    fingerprint = {'bytes': len(raw), 'sha256': digest(raw)}
    assert len(raw) <= OUTPUT_CAP
    if pin is not None:
        assert fingerprint == pin
    if rel in read_fingerprints:
        assert read_fingerprints[rel] == fingerprint
    read_fingerprints[rel] = fingerprint
    return raw

def owned_root_inventory():

        'result/shared_f4_PARTITION.json', 'result/capable_single_PARTITION.json', 'result/untied4_PARTITION.json'}
    paths = sorted([ROOT, *ROOT.rglob('*')])
    assert len(paths) <= 512
    inventory, total = [], 0
    for path in paths:
        confined(path)
        rel = path.relative_to(ROOT).as_posix()
        if path.is_dir():
            inventory.append({'path': rel, 'type': 'directory', 'bytes': 0, 'sha256': None})
        else:
            assert path.is_file() and rel in allowed
            assert total + path.stat().st_size <= OUTPUT_CAP
            checked_bytes(rel)
            total += read_fingerprints[rel]['bytes']
            assert total <= OUTPUT_CAP
            inventory.append({'path': rel, 'type': 'ordinary_file', **read_fingerprints[rel]})
    # A complete bounded inventory rejects unknown payload paths before hashing.
    for entry in inventory:
        if entry['type'] == 'ordinary_file':
            checked_bytes(entry['path'], {key: entry[key] for key in ('bytes', 'sha256')})
    assert sorted([ROOT, *ROOT.rglob('*')]) == paths
    return inventory, total
